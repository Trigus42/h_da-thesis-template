\newglossaryentry{clt}{
    name={Central Limit Theorem},
    description={A fundamental theorem in probability theory that establishes that the sum of many independent and identically distributed random variables tends toward a normal distribution, regardless of the original distribution},
    first={Central Limit Theorem (CLT)\glsadd{clt}}
}

\newglossaryentry{normal}{
    name={Normal Distribution},
    description={A continuous probability distribution that has a bell-shaped probability density function, also known as the Gaussian distribution},
    symbol={$N(\mu,\sigma^2)$}
}

\newglossaryentry{iid}{
    name={i.i.d.},
    description={Independent and identically distributed random variables; a sequence of random variables that are independent of each other and have the same probability distribution},
    first={independent and identically distributed (i.i.d.)\glsadd{iid}}
}

\newglossaryentry{variance}{
    name={Variance},
    description={A measure of dispersion in a probability distribution, showing how far values are spread out from the average},
    symbol={$\sigma^2$}
}

\newglossaryentry{pmf}{
    name={PMF},
    description={Probability Mass Function; a function that gives the probability that a discrete random variable is exactly equal to some value},
    first={Probability Mass Function (PMF)\glsadd{pmf}}
}
